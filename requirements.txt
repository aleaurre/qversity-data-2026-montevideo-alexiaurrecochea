# Python dependencies for the Airflow container.
#
# Pinning rationale:
#   - >= X, < Y  : pin to a minor release line. Patch versions roll forward
#                  (security fixes) but no minor surprises.
#   - == X.Y.Z   : pin exact only when we know about a breaking change in
#                  the next patch.
#
# Order: first the Airflow extras (if any), then data tooling, then dbt.

# ─── PySpark (Silver layer transformations) ─────────────────────────────────
# 3.5.x is the line the project spec requires ("PySpark 3.5+"). 3.5 is also
# the last line that supports Python 3.11 without issues; 4.x dropped 3.11.
pyspark>=3.5.0,<3.6.0

# ─── Postgres connectivity ──────────────────────────────────────────────────
# psycopg2-binary is fine for dev. In prod you'd build psycopg2 from source
# against the system libpq; for this project the binary wheel is enough.
psycopg2-binary>=2.9.9,<2.10.0

# ─── AWS S3 access for the bronze ingestion ─────────────────────────────────
# Used by the DAG to download fintech_banking_dataset.json. Even though the
# bucket is public, boto3 is more reliable than raw urllib for retries and
# error handling.
boto3>=1.34.0,<2.0.0

# ─── dbt (Silver normalization + Gold analytics) ────────────────────────────
# Spec says "dbt-core + dbt-postgres 1.7+". We pin to the 1.7.x line for
# build reproducibility. Both packages must be on the same minor version.
dbt-core>=1.7.0,<1.8.0
dbt-postgres>=1.7.0,<1.8.0

# want on day 6+. Install via `dbt deps` (it's a dbt package, not Python),
# so it goes in packages.yml, not here. Listed here as a reminder only.
